```javascript
// 修复前：累加后除以数量
w.c += (+r['作业完成率'] || 0) * 100;
w.c = w.n ? (d.c / d.n).toFixed(1) : 0;

// 修复后：明确计算平均值
w.cSum += (+r['作业完成率'] || 0) * 100;  // 累加

w.c = w.n ? (w.cSum / w.n).toFixed(1) : 0;  // 平均值
```

**图表优化：**

- 完成率 Y 轴限制 0-100%

## 📊 新表格结构详解

### 表头结构

```
┌─────────────────────────────────────────────────────────────────────────┐
│ 省份 │ 城市 │ 区县 │ 学校 │ 年级 │ 班级 │  第 1 周  │  第 2 周  │ ... │ 未过期 │
│      │      │      │      │      │      │ 布置│完成│ 布置│完成│     │ 付费│试用│
├─────────────────────────────────────────────────────────────────────────┤
│ 湖北 │ 孝感 │ 汉川 │ XX 中 │ 九 1 │ 1 班  │ 10 │85%│ 12 │90%│ ... │  45 │  5 │
└─────────────────────────────────────────────────────────────────────────┘
```

### 列说明

#### 固定列（左侧）
1. **省份** - 学校所在省份
2. **城市** - 学校所在城市
3. **区县** - 学校所在区县
4. **学校** - 学校名称
5. **年级** - 年级（如：七年级、八年级、九年级）
6. **班级** - 班级名称

#### 动态列（按周展示）
每个上传的 Excel 文件对应 2 列：
- **布置次数** - 该周布置作业的总次数
- **完成率** - 该周作业完成率的平均值

**示例：**
- 上传 1 个文件 → 表格 +2 列
- 上传 2 个文件 → 表格 +4 列
- 上传 5 个文件 → 表格 +10 列

#### 统计列（右侧末尾）
- **未过期付费学生数** - 仅显示最后一周的数据
- **未过期试用学生数** - 仅显示最后一周的数据

## 🔧 技术实现

### 数据聚合逻辑

```javascript
// 按学校、年级、班级分组
const groupMap = new Map();
AppState.filteredData.forEach(r => {
    const key = `${r['省份']}|${r['城市']}|${r['学校名称']}|${r['年级']}|${r['班级名称']}`;
    
    // 同一班级的多周数据聚合到一行
    if (!groupMap.has(key)) { /* 创建分组 */ }
    
    const g = groupMap.get(key);
    const weekKey = r.weekLabel;
    
    // 按周次聚合
    if (!g.weeks.has(weekKey)) { /* 创建周次数据 */ }
    
    const w = g.weeks.get(weekKey);
    w.assignments += r['布置作业次数'];      // 累加
    w.completionSum += r['作业完成率'] * 100; // 累加
    w.count++;
    w.completionRate = w.count ? (w.completionSum / w.count) : 0;  // 平均
});
```

### 表头生成

```javascript
// 动态生成表头
let thead = '<tr>';
// 固定列
thead += '<th rowspan="2">省份</th>...<th rowspan="2">班级</th>';

// 动态周列
sortedWeeks.forEach(week => {
    thead += `<th colspan="2">${week 的日期范围}</th>`;
});

// 统计列
thead += '<th rowspan="2">未过期付费学生数</th>';
thead += '<th rowspan="2">未过期试用学生数</th>';
thead += '</tr>';

// 第二行表头（周次细分）
sortedWeeks.forEach(() => {
    thead += '<th>布置次数</th><th>完成率</th>';
});
```

### 数据行生成

```javascript
sortedGroups.forEach(g => {
    let row = `<tr>${g.province}${g.city}${g.school}${g.grade}${g.className}`;
    
    // 每周数据
    sortedWeeks.forEach(weekKey => {
        if (g.weeks.has(weekKey)) {
            const w = g.weeks.get(weekKey);
            row += `<td>${w.assignments}</td><td>${w.completionRate}%</td>`;
        } else {
            row += '<td>-</td><td>-</td>';  // 该周无数据
        }
    });
    
    // 最后一周的统计
    const lastWeek = g.weeks.get(lastWeekKey);
    row += `<td>${lastWeek.paidCount}</td><td>${lastWeek.trialCount}</td>`;
    row += '</tr>';
});
```

## ⚠️ 注意事项

### 数据展示规则

1. **布置次数**：该周累加值
2. **完成率**：该周平均值（更科学）
3. **未过期学生数**：仅显示最后一周（避免重复）

### 空数据展示

- 某班级某周无数据 → 显示"-"
- 所有列宽自适应，可横向滚动

### 导出 CSV

- 导出格式与表格一致
- 包含所有周次数据
- 适合用 Excel 进一步分析
